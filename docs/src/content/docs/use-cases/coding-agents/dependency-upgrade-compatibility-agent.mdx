## What you are building

This example upgrades a concrete dependency in a deterministic sample workspace that lives inside a cloned public repository. It upgrades the dependency, runs the sample test suite, and exports a patch only when the upgraded workspace still passes validation.

## Why AerolVM fits

- The upgrade run happens away from shared CI and developer laptops.
- A deterministic sample workspace removes guesswork from the example.
- The final patch and summary can be downloaded directly into a review system.

## Copy-paste TypeScript script

This script is self-contained. The only required environment variable is `SB_PAT_TOKEN`. If `SB_API_URL` is unset, the script defaults to `http://127.0.0.1:8080`.

```ts
import { MicroVM } from "@aerol-ai/aerolvm-sdk";
import { writeFile } from "node:fs/promises";

const apiUrl = process.env.SB_API_URL ?? "http://127.0.0.1:8080";
const patToken = process.env.SB_PAT_TOKEN;

if (!patToken) {
  throw new Error("Set SB_PAT_TOKEN before running this example.");
}

const repoURL = "https://github.com/axios/axios.git";
const samplePackageJSON = JSON.stringify(
  {
    name: "aerolvm-upgrade-example",
    private: true,
    version: "1.0.0",
    type: "module",
    scripts: {
      test: "node test.mjs",
    },
    dependencies: {
      chalk: "4.1.2",
    },
  },
  null,
  2,
);

const sampleTest = `import assert from "node:assert/strict";
import chalk from "chalk";

const rendered = chalk.green("upgrade-ok");
assert.equal(typeof rendered, "string");
console.log("dependency smoke test passed");
`;

async function main() {
  const client = new MicroVM({ apiUrl, patToken });
  const sandbox = await client.create({
    image: "node:22-bookworm",
    cpu: 2,
    memoryMB: 4096,
    diskGB: 18,
    containerCommand: ["sh", "-lc", "while true; do sleep 3600; done"],
  });

  try {
    await sandbox.exec("apt-get update && apt-get install -y git");
    await sandbox.exec("mkdir -p /workspace/out");
    await sandbox.exec({
      command: `git clone --depth=1 ${repoURL} /workspace/app`,
    });
    await sandbox.exec("mkdir -p /workspace/app/examples/aerolvm-upgrade");

    await sandbox.uploadFile(
      "/workspace/app/examples/aerolvm-upgrade/package.json",
      samplePackageJSON,
    );
    await sandbox.uploadFile(
      "/workspace/app/examples/aerolvm-upgrade/test.mjs",
      sampleTest,
    );

    const run = sandbox.execStream({
      command: [
        "npm install",
        "npx --yes npm-check-updates -u",
        "npm install",
        "npm test",
        "printf '%s\\n' 'Workspace: examples/aerolvm-upgrade' 'Action: upgraded chalk to the latest version' 'Validation: npm test passed' > /workspace/out/upgrade-summary.txt",
        "git diff -- package.json package-lock.json > /workspace/out/upgrade.patch",
      ].join(" && "),
      workdir: "/workspace/app/examples/aerolvm-upgrade",
      onStdout: (chunk) => process.stdout.write(chunk),
      onStderr: (chunk) => process.stderr.write(chunk),
    });

    const exit = await run.done;
    if (exit.code !== 0) {
      throw new Error(`upgrade run failed with exit code ${exit.code}`);
    }

    const decoder = new TextDecoder();
    const summary = decoder.decode(
      await sandbox.downloadFile("/workspace/out/upgrade-summary.txt"),
    );
    const patch = decoder.decode(
      await sandbox.downloadFile("/workspace/out/upgrade.patch"),
    );

    await writeFile("dependency-upgrade-summary.txt", summary);
    await writeFile("dependency-upgrade.patch", patch);

    console.log(
      JSON.stringify(
        {
          sandboxID: sandbox.id,
          summaryFile: "dependency-upgrade-summary.txt",
          patchFile: "dependency-upgrade.patch",
        },
        null,
        2,
      ),
    );
  } finally {
    await sandbox.destroy();
  }
}

main().catch((error) => {
  console.error(error);
  process.exitCode = 1;
});
```

## Deliverables

- `dependency-upgrade.patch` for review.
- `dependency-upgrade-summary.txt` with the exact validation result.
