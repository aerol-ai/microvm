## What you are building

This workflow takes generated application files, boots them inside a sandbox, streams the build output, and returns a live preview URL plus a built artifact that can be shown directly in the product.

## Why AerolVM fits

- The generated code runs away from the host and shared web infrastructure.
- Streamed install and build logs can be reflected into the builder UI in real time.
- The same runtime can expose previews, export artifacts, and later be cleaned up by lifecycle policy.

## Copy-paste TypeScript script

This script is self-contained. The only required environment variable is `SB_PAT_TOKEN`. If `SB_API_URL` is unset, the script defaults to `http://127.0.0.1:8080`.

```ts
import { MicroVM } from "@aerol-ai/aerolvm-sdk";
import { writeFile } from "node:fs/promises";

const apiUrl = process.env.SB_API_URL ?? "http://127.0.0.1:8080";
const patToken = process.env.SB_PAT_TOKEN;

if (!patToken) {
  throw new Error("Set SB_PAT_TOKEN before running this example.");
}

const generatedFiles = [
  {
    path: "package.json",
    contents: JSON.stringify(
      {
        name: "aerolvm-generated-app",
        private: true,
        version: "1.0.0",
        type: "module",
        scripts: {
          build: "node build.mjs",
          dev: "node server.mjs",
        },
      },
      null,
      2,
    ),
  },
  {
    path: "build.mjs",
    contents: `import { mkdir, readFile, writeFile } from "node:fs/promises";
await mkdir("dist", { recursive: true });
const html = await readFile("src/index.html", "utf8");
await writeFile("dist/index.html", html);
console.log("build complete");
`,
  },
  {
    path: "server.mjs",
    contents: `import { createServer } from "node:http";
import { readFile } from "node:fs/promises";

const html = await readFile("dist/index.html", "utf8");
const port = Number(process.env.PORT || 3000);

createServer((_request, response) => {
  response.writeHead(200, { "content-type": "text/html; charset=utf-8" });
  response.end(html);
}).listen(port, "0.0.0.0", () => {
  console.log("preview server listening on", port);
});
`,
  },
  {
    path: "src/index.html",
    contents: `<!doctype html>
<html lang="en">
  <head>
    <meta charset="UTF-8" />
    <meta name="viewport" content="width=device-width, initial-scale=1.0" />
    <title>Generated AerolVM App</title>
  </head>
  <body>
    <main>
      <h1>Generated app preview</h1>
      <p>This page was built from generated files inside AerolVM.</p>
    </main>
  </body>
</html>
`,
  },
];

async function main() {
  const client = new MicroVM({ apiUrl, patToken });
  const sandbox = await client.create({
    image: "node:22-bookworm",
    cpu: 2,
    memoryMB: 4096,
    diskGB: 20,
    containerCommand: ["sh", "-lc", "while true; do sleep 3600; done"],
    lifecycle: {
      destroyIfIdleFor: 45 * 60 * 1_000_000_000,
    },
  });

  await sandbox.exec("mkdir -p /workspace/generated-app/src /workspace/out");
  await Promise.all(
    generatedFiles.map((file) =>
      sandbox.uploadFile(`/workspace/generated-app/${file.path}`, file.contents),
    ),
  );

  const buildLogs: string[] = [];
  const capture = (chunk: string) => {
    buildLogs.push(chunk);
    process.stdout.write(chunk);
  };

  const install = sandbox.execStream({
    command: "npm install && npm run build",
    workdir: "/workspace/generated-app",
    onStdout: capture,
    onStderr: capture,
  });

  const installExit = await install.done;
  if (installExit.code !== 0) {
    throw new Error(`builder bootstrap failed with exit code ${installExit.code}`);
  }

  const previewSession = await sandbox.createSession({
    name: "builder-preview",
    command: "npm run dev",
    workDir: "/workspace/generated-app",
    env: { PORT: "3000" },
  });

  await new Promise((resolve) => setTimeout(resolve, 2000));
  const previewURL = await sandbox.exposePort(3000);
  const builtHTML = new TextDecoder().decode(
    await sandbox.downloadFile("/workspace/generated-app/dist/index.html"),
  );

  await writeFile("ai-app-builder-runtime.log", buildLogs.join(""));
  await writeFile("ai-app-builder-runtime-built-index.html", builtHTML);
  await writeFile(
    "ai-app-builder-runtime.json",
    `${JSON.stringify(
      {
        sandboxID: sandbox.id,
        previewSessionID: previewSession.id,
        previewURL,
      },
      null,
      2,
    )}\n`,
  );

  console.log(
    JSON.stringify(
      {
        sandboxID: sandbox.id,
        previewSessionID: previewSession.id,
        previewURL,
      },
      null,
      2,
    ),
  );
}

main().catch((error) => {
  console.error(error);
  process.exitCode = 1;
});
```

## Product outputs

- `ai-app-builder-runtime.log` with streamed build output.
- `ai-app-builder-runtime-built-index.html` as the exported artifact.
- `ai-app-builder-runtime.json` with the preview details.
